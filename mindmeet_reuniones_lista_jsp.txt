    <script>
        // Función para filtrar reuniones
        function filtrarReuniones() {
            const searchInput = document.getElementById('searchInput').value.toLowerCase();
            const cards = document.querySelectorAll('.reunion-card');
            
            cards.forEach(card => {
                const titulo = card.getAttribute('data-titulo').toLowerCase();
                if (titulo.includes(searchInput)) {
                    card.style.display = 'block';
                } else {
                    card.style.display = 'none';
                }
            });
        }
        
        // Función para confirmar eliminación
        function confirmarEliminacion(id, titulo) {
            if (confirm('¿Está seguro que desea eliminar la reunión "' + titulo + '"?')) {
                window.location.href = '${pageContext.request.contextPath}/reunion?action=eliminar&id=' + id;
            }
        }
    </script>
